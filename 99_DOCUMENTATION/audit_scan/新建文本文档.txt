import os,json

root=r"E:\football_v"

keywords=[
"MODEL_LAYER",
"model_layer",
"elo_model",
"dixon_coles",
"poisson",
"xgboost",
"model_registry",
"model_store.db"
]


result=[]

for path,dirs,files in os.walk(root):

    dirs[:]=[
        d for d in dirs
        if d not in [
            "__pycache__",
            ".git",
            ".pytest_cache"
        ]
    ]

            pass

        ensure_ascii=False
    )
